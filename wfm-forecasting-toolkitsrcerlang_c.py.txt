"""
Erlang C Calculation Module
Industry-Standard Workforce Management Formulas
Production-Ready Mathematical Functions
"""

import math
from typing import Dict, Tuple
from scipy.special import factorial

class ErlangCCalculator:
    """Encapsulates Erlang C calculations for WFM"""
    
    @staticmethod
    def erlang_c(agents: float, traffic_intensity: float) -> float:
        """
        Calculate Erlang C probability (probability of queueing).
        
        FORMULA:
        Erlang C = [A^N / N! × N/(N-A)] / [Σ(A^k/k!) + A^N/N! × N/(N-A)]
        
        Where:
        A = Traffic Intensity (Erlangs) = Call Volume × AHT / Interval Duration
        N = Number of agents
        
        Args:
            agents: Number of available agents
            traffic_intensity: Traffic in Erlangs (call volume × AHT / 3600)
            
        Returns:
            float: Probability that a call waits in queue (0-1)
            
        Raises:
            ValueError: If agents <= 0 or traffic >= agents
            
        Example:
            >>> pw = ErlangCCalculator.erlang_c(10, 8.5)
            >>> print(f"{pw:.4f}")  # 0.3847 = 38.47% wait
        """
        
        # Input validation
        if agents <= 0:
            raise ValueError("Agents must be > 0")
        if traffic_intensity < 0:
            raise ValueError("Traffic intensity must be >= 0")
        
        # Edge case: no traffic
        if traffic_intensity == 0:
            return 0.0
        
        # Edge case: system overloaded
        if traffic_intensity >= agents:
            return 1.0
        
        try:
            # Numerator: A^N / N! × N/(N-A)
            numerator = (math.pow(traffic_intensity, agents) / factorial(agents)) * \
                       (agents / (agents - traffic_intensity))
            
            # Denominator sum
            denominator_sum = 0
            
            # Sum from k=0 to N-1: A^k / k!
            for k in range(int(agents)):
                denominator_sum += math.pow(traffic_intensity, k) / factorial(k)
            
            # Add final term: A^N / N! × N/(N-A)
            denominator_sum += (math.pow(traffic_intensity, agents) / factorial(agents)) * \
                              (agents / (agents - traffic_intensity))
            
            # Calculate Erlang C
            erlang_c_value = numerator / denominator_sum
            
            # Ensure value is between 0 and 1
            return min(erlang_c_value, 1.0)
        
        except Exception as e:
            raise ValueError(f"Erlang C calculation error: {str(e)}")
    
    @staticmethod
    def service_level(agents: int, traffic_intensity: float, 
                     target_answer_time: float, aht: float) -> float:
        """
        Calculate service level (% of calls answered within target time).
        
        FORMULA:
        SL = 1 - [Pw × e^(-(N-A)(t/AHT))]
        
        Where:
        Pw = Erlang C value
        N = Agents
        A = Traffic Intensity
        t = Target Answer Time
        AHT = Average Handle Time
        
        Args:
            agents: Number of agents
            traffic_intensity: Traffic in Erlangs
            target_answer_time: Target answer time in seconds
            aht: Average Handle Time in seconds
            
        Returns:
            float: Service level percentage (0-100)
            
        Example:
            >>> sl = ErlangCCalculator.service_level(12, 8.5, 20, 180)
            >>> print(f"{sl:.2f}%")  # 89.23%
        """
        
        if aht <= 0:
            raise ValueError("AHT must be > 0")
        
        # Get Erlang C probability
        pw = ErlangCCalculator.erlang_c(agents, traffic_intensity)
        
        # If no queueing probability, SL is 100%
        if pw == 0:
            return 100.0
        
        # Calculate exponent: -(N-A) × (t/AHT)
        exponent = -(agents - traffic_intensity) * (target_answer_time / aht)
        
        try:
            exponential_term = math.exp(exponent)
            sl = 1 - (pw * exponential_term)
            return max(0, min(100, sl * 100))
        except Exception as e:
            return 0.0
    
    @staticmethod
    def average_speed_of_answer(agents: int, traffic_intensity: float, aht: float) -> float:
        """
        Calculate Average Speed of Answer (ASA) in seconds.
        
        FORMULA:
        ASA = (Pw × AHT) / (N - A)
        
        Args:
            agents: Number of agents
            traffic_intensity: Traffic in Erlangs
            aht: Average Handle Time in seconds
            
        Returns:
            float: ASA in seconds
            
        Example:
            >>> asa = ErlangCCalculator.average_speed_of_answer(12, 8.5, 180)
            >>> print(f"{asa:.1f} seconds")  # 45.2 seconds
        """
        
        # Avoid division by zero
        if agents <= traffic_intensity:
            return float('inf')
        
        pw = ErlangCCalculator.erlang_c(agents, traffic_intensity)
        asa = (pw * aht) / (agents - traffic_intensity)
        
        return max(0, asa)
    
    @staticmethod
    def occupancy(traffic_intensity: float, agents: int) -> float:
        """
        Calculate agent occupancy percentage.
        
        FORMULA:
        Occupancy = (A / N) × 100
        
        Args:
            traffic_intensity: Traffic in Erlangs
            agents: Number of agents
            
        Returns:
            float: Occupancy percentage (0-100)
            
        Example:
            >>> occ = ErlangCCalculator.occupancy(8.5, 12)
            >>> print(f"{occ:.2f}%")  # 70.83%
        """
        
        if agents == 0:
            return 0
        
        occ = (traffic_intensity / agents) * 100
        return min(occ, 100)
    
    @staticmethod
    def required_agents(call_volume: int, aht: float, interval_minutes: float,
                       target_sl: float, target_answer_time: float,
                       max_occupancy: float = 85) -> Dict:
        """
        Calculate required number of agents to meet service level target.
        
        Args:
            call_volume: Number of calls in the interval
            aht: Average Handle Time in seconds
            interval_minutes: Length of interval (typically 15, 30, or 60)
            target_sl: Target service level % (e.g., 80 for 80/20 rule)
            target_answer_time: Target answer time in seconds (e.g., 20)
            max_occupancy: Maximum acceptable occupancy % (default 85)
            
        Returns:
            dict with:
            - required_agents (int)
            - achieved_sl (float)
            - occupancy (float)
            - asa (float)
            - traffic_intensity (float)
            
        Example:
            >>> result = ErlangCCalculator.required_agents(
            ...     call_volume=100,
            ...     aht=180,
            ...     interval_minutes=30,
            ...     target_sl=80,
            ...     target_answer_time=20
            ... )
            >>> print(f"Need {result['required_agents']} agents")
        """
        
        # Validate inputs
        if call_volume < 0 or aht <= 0 or interval_minutes <= 0:
            raise ValueError("Call volume, AHT, and interval must be positive")
        
        # Calculate traffic intensity in Erlangs
        interval_seconds = interval_minutes * 60
        traffic_intensity = (call_volume * aht) / interval_seconds
        
        # Start with minimum agents (ceiling of traffic intensity)
        agents = max(1, math.ceil(traffic_intensity))
        
        # Increment until we meet both SL and occupancy targets
        max_iterations = 1000
        for i in range(max_iterations):
            current_sl = ErlangCCalculator.service_level(agents, traffic_intensity,
                                                        target_answer_time, aht)
            current_occ = ErlangCCalculator.occupancy(traffic_intensity, agents)
            
            # Check if targets met
            if current_sl >= target_sl and current_occ <= max_occupancy:
                asa = ErlangCCalculator.average_speed_of_answer(agents, traffic_intensity, aht)
                
                return {
                    'required_agents': agents,
                    'achieved_sl': round(current_sl, 2),
                    'occupancy': round(current_occ, 2),
                    'asa': round(asa, 2),
                    'traffic_intensity': round(traffic_intensity, 2)
                }
            
            agents += 1
        
        # If we max out iterations, return best effort
        asa = ErlangCCalculator.average_speed_of_answer(agents, traffic_intensity, aht)
        return {
            'required_agents': agents,
            'achieved_sl': round(ErlangCCalculator.service_level(agents, traffic_intensity,
                                                                target_answer_time, aht), 2),
            'occupancy': round(ErlangCCalculator.occupancy(traffic_intensity, agents), 2),
            'asa': round(asa, 2),
            'traffic_intensity': round(traffic_intensity, 2),
            'warning': 'Could not meet targets within reasonable agent count'
        }
